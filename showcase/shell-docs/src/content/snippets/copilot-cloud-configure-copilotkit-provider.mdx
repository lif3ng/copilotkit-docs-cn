[`<CopilotKit>`](/reference/v1/components/CopilotKit) 提供方必须包裹应用中使用 Copilot 功能的部分。
对于大多数场景，直接用 `CopilotKit` 提供方包裹整个应用即可，例如放在你的 `layout.tsx` 中。

<Callout type="info">
  注意：`<CopilotKit>` 提供方可以添加到应用中的任意位置。实际上，如果你想要多个相互独立的 copilot，甚至可以在一个应用中使用多个 `<CopilotKit>` 提供方。
</Callout>

<Callout type="info">
    <LinkToCopilotCloud asButton={false}>点击此处免费获取你的 Copilot Cloud API 密钥</LinkToCopilotCloud>。然后，将 `<your-public-api-key>` 替换为你实际的 API 密钥。
</Callout>

```tsx title="layout.tsx" showLineNumbers
import "./globals.css";

import { ReactNode } from "react";
import { CopilotKit } from "@copilotkit/react-core/v2"; // [!code highlight]

export default function RootLayout({ children }: { children: ReactNode }) {
  return (
    <html lang="en">
      <body>
        {/* Use the public api key you got from Copilot Cloud  */}
        {/* [!code highlight:3] */}
        <CopilotKit publicApiKey="<your-copilot-cloud-public-api-key>">
          {children}
        </CopilotKit>
      </body>
    </html>
  );
}
```
